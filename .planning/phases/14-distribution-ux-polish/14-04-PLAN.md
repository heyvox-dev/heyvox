Wire the wake-word model download into the setup wizard (D-19) and lock the D-18 co-default fallback (`wake_words.also_load: [hey_jarvis_v0.1]`) with a regression test. This plan converts the artifact + constants produced in Plan 14-03 into an end-user experience: a fresh `pip install heyvox && heyvox setup` brings down `hey_vox.onnx` to `~/.config/heyvox/models/`, verifies integrity, and falls back gracefully to `hey_jarvis_v0.1` if anything fails.

Purpose: SPEC-R6 acceptance criterion #7 says fresh install + `heyvox setup` must download the general model AND `heyvox start` must recognize "Hey Vox". The download flow has three failure modes that must each be handled correctly:
1. Network failure → daemon stays functional via `hey_jarvis_v0.1` co-default
2. sha256 mismatch → temp file deleted, existing model (if any) preserved, wizard reports failure but continues
3. Re-run of `heyvox setup` after success → existing `hey_vox.onnx` (potentially a user-trained personalised model) is preserved; only `--redownload-wakeword` forces refresh

The D-18 co-default (`also_load: [hey_jarvis_v0.1]`) is ALREADY in code at `heyvox/config.py:54` per RESEARCH.md finding #6. This plan does NOT add the default — it adds a regression test so accidental future edits trip CI.

    ```python
    """Unit tests for the wake-word model download step in heyvox.setup.wizard.

    Phase 14 / SPEC R6 / D-19. Mocks urllib.request.urlopen + hashlib so tests
    don't need network access and don't depend on the real GitHub Releases asset.

    NOTE: monkeypatches target the SOURCE modules (heyvox.constants,
    heyvox.config) NOT the consumer module (heyvox.setup.wizard) because the
    helper uses function-local imports that re-resolve on each call. See
    user-memory feedback_function_local_import_patching.md.
    """

    from __future__ import annotations

    import hashlib
    from pathlib import Path
    from unittest.mock import MagicMock, patch


    def _make_mock_response(content: bytes):
        """Build a fake urlopen response for use as patch return_value."""
        resp = MagicMock()
        resp.headers.get.return_value = str(len(content))
        # Yield content in 64KB chunks, terminated by an empty chunk
        chunks = [content[i : i + 64 * 1024] for i in range(0, len(content), 64 * 1024)] + [b""]
        resp.read.side_effect = chunks
        resp.__enter__.return_value = resp
        resp.__exit__.return_value = False
        return resp


    def test_download_writes_file_with_correct_sha256(tmp_path, monkeypatch):
        from heyvox.setup import wizard

        content = b"FAKE_ONNX_BYTES_" * 1000  # ~17 KB
        expected_sha = hashlib.sha256(content).hexdigest()

        monkeypatch.setattr("heyvox.constants.HEY_VOX_MODEL_URL", "https://example.com/hey_vox.onnx")
        monkeypatch.setattr("heyvox.constants.HEY_VOX_MODEL_SHA256", expected_sha)
        monkeypatch.setattr("heyvox.config.CONFIG_DIR", tmp_path)

        console = MagicMock()
        with patch("urllib.request.urlopen", return_value=_make_mock_response(content)):
            ok = wizard._download_wakeword_model(console, force=False)

        assert ok is True
        target = tmp_path / "models" / "hey_vox.onnx"
        assert target.exists()
        assert target.read_bytes() == content


    def test_sha256_mismatch_aborts_and_returns_false(tmp_path, monkeypatch):
        """If sha256 doesn't match, the helper unlinks the temp file and returns False."""
        from heyvox.setup import wizard

        content = b"WRONG_BYTES" * 100
        monkeypatch.setattr("heyvox.constants.HEY_VOX_MODEL_URL", "https://example.com/hey_vox.onnx")
        monkeypatch.setattr("heyvox.constants.HEY_VOX_MODEL_SHA256", "0" * 64)  # cannot match real sha256
        monkeypatch.setattr("heyvox.config.CONFIG_DIR", tmp_path)

        console = MagicMock()
        with patch("urllib.request.urlopen", return_value=_make_mock_response(content)):
            ok = wizard._download_wakeword_model(console, force=False)

        assert ok is False
        target = tmp_path / "models" / "hey_vox.onnx"
        assert not target.exists()
        # No temp file leak either — check the models/ dir is empty (or only has the dir itself)
        models_dir = tmp_path / "models"
        if models_dir.exists():
            assert list(models_dir.iterdir()) == []

        from heyvox.setup import wizard

        monkeypatch.setattr("heyvox.config.CONFIG_DIR", tmp_path)
    ```

    ```python
    def _download_wakeword_model(console, force: bool = False) -> bool:
        """Download hey_vox.onnx from GitHub Releases to ~/.config/heyvox/models/.

        Phase 14 / SPEC R6 / D-19.

        Args:
            console: rich.Console for output.
            force: If True, re-download even if file exists.

        Returns:
            True if model is present after this call (downloaded or already there).
            False if download failed (network / sha mismatch). Wizard continues
            either way — hey_jarvis_v0.1 co-default keeps the daemon functional.
        """
        # Function-local imports so monkeypatched constants are re-resolved on
        # each call (per tests/test_setup_wakeword_download.py).
        import hashlib
        import os
        import tempfile
        import urllib.request

        from heyvox.config import CONFIG_DIR
        from heyvox.constants import HEY_VOX_MODEL_URL, HEY_VOX_MODEL_SHA256
        from rich.progress import (
            BarColumn,
            DownloadColumn,
            Progress,
            TextColumn,
            TransferSpeedColumn,
        )

        models_dir = CONFIG_DIR / "models"
        models_dir.mkdir(parents=True, exist_ok=True)
        target = models_dir / "hey_vox.onnx"

        if target.exists() and not force:
            console.print(f"  [green]✓[/green] hey_vox model present: {target}")
            return True

        tmp_path = None
        try:
            with urllib.request.urlopen(HEY_VOX_MODEL_URL) as resp:
                total = int(resp.headers.get("Content-Length", 0) or 0)
                with tempfile.NamedTemporaryFile(
                    delete=False, suffix=".onnx", dir=str(models_dir),
                ) as tmp:
                    hasher = hashlib.sha256()
                    with Progress(
                        TextColumn("[progress.description]{task.description}"),
                        BarColumn(),
                        DownloadColumn(),
                        TransferSpeedColumn(),
                        console=console,
                    ) as progress:
                        task = progress.add_task("Downloading hey_vox.onnx", total=total or None)
                        while True:
                            chunk = resp.read(64 * 1024)
                            if not chunk:
                                break
                            tmp.write(chunk)
                            hasher.update(chunk)
                            progress.update(task, advance=len(chunk))
                    tmp_path = tmp.name

            actual = hasher.hexdigest()
            if actual.lower() != HEY_VOX_MODEL_SHA256.lower():
                if tmp_path:
                    try:
                        os.unlink(tmp_path)
                    except OSError:
                        pass
                console.print(
                    f"  [red]✗[/red] sha256 mismatch: "
                    f"expected {HEY_VOX_MODEL_SHA256[:12]}…, got {actual[:12]}…"
                )
                return False

            os.replace(tmp_path, target)
            console.print(f"  [green]✓[/green] hey_vox model downloaded: {target}")
            return True

        except Exception as e:  # noqa: BLE001 — graceful-fail per D-19
            # Network failure / sha mismatch / OS error — leave any existing
            # target untouched, clean up the temp file, log a soft warning.
            if tmp_path:
                try:
                    os.unlink(tmp_path)
                except OSError:
                    pass
            console.print(f"  [yellow]![/yellow] Download failed: {e}")
            console.print(
                "  [dim]The bundled hey_jarvis_v0.1 fallback will be used until "
                "you retry with `heyvox setup --redownload-wakeword`.[/dim]"
            )
            return False
    ```
